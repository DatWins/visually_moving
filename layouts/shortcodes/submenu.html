{{- /* Renders a horizontal pill row of sub-items within a section. Usage: {{< submenu "Label1|anchor1" "Label2|anchor2" >}} Items without a "|" separator are shown as "coming soon" (dashed outline). */ -}}
{{- range .Params -}} {{- $parts := split . "|" -}} {{- if gt (len $parts) 1 -}} {{ index $parts 0 }} {{- else -}} {{ index $parts 0 }} {{- end -}} {{- end -}}
